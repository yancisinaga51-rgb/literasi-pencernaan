Sinkronisasi Opsional ke Google Sheets

Teruskan otomatis setiap nilai siswa ke Google Spreadsheet pribadi Anda.

Langkah 1: Buka Google Sheets baru → klik menu Ekstensi (Extensions) > Apps Script, lalu tempel kode berikut dan Deploy sebagai Web App (Akses: Siapa saja / Anyone):

function doPost(e) {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  if (sheet.getLastRow() === 0) {
    sheet.appendRow(["Waktu Submit", "Nama Siswa", "Kelas", "Skor", "Benar", "Salah", "Kosong", "Durasi"]);
  }
  var data = JSON.parse(e.postData.contents);
  sheet.appendRow([
    data.submittedAtText,
    data.nama,
    data.kelas,
    data.skor,
    data.benar,
    data.salah,
    data.kosong,
    data.durasiText
  ]);
  return ContentService.createTextOutput(JSON.stringify({status: "ok"})).setMimeType(ContentService.MimeType.JSON);
}
